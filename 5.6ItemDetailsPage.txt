import React, { useEffect, useState } from "react";
import { api } from "../api/api";
import { useAuth } from "../context/AuthContext";
import ChatBox from "../components/ChatBox";

export default function ItemDetailPage({ itemId, onBack }) {
  const { token } = useAuth();
  const [item, setItem] = useState(null);

  useEffect(() => {
    async function load() {
      const data = await api.getItem(itemId);
      setItem(data);
    }
    load();
  }, [itemId, token]);

  if (!item) return <div className="page">Loading...</div>;

  const imgSrc = item.imagePath ? `http://localhost:4000/uploads/${item.imagePath}` : null;

  return (
    <div className="page">
      <button className="secondary" onClick={onBack}>&larr; Back</button>

      <div className="card">
        {imgSrc ? <img className="hero" src={imgSrc} alt={item.title} /> : null}
        <h2>{item.title}</h2>
        <div className="muted">{item.category || "Uncategorised"} • {item.itemCondition || "N/A"}</div>
        <p>{item.description}</p>
        <div className="muted">Owner: {item.owner?.displayName} ({item.owner?.email})</div>
      </div>

      <ChatBox itemId={item.id} />
    </div>
  );
}
