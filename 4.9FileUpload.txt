// fileUpload.js
// No multer. We accept an image as base64 DataURL from the frontend.
// We decode it and save into /uploads, then store imagePath in DB.

const fs = require("fs");
const path = require("path");

function ensureDir(dir) {
  if (!fs.existsSync(dir)) fs.mkdirSync(dir, { recursive: true });
}

// input: "data:image/png;base64,iVBORw0KGgoAAA...."
function saveDataUrlImage(dataUrl, uploadRoot) {
  if (!dataUrl) return null;

  const match = dataUrl.match(/^data:(image\/\w+);base64,(.+)$/);
  if (!match) throw new Error("Invalid image format. Please upload a valid image.");

  const mime = match[1];           // e.g. image/png
  const base64 = match[2];
  const ext = mime.split("/")[1];  // png, jpeg, etc.

  const fileName = `item_${Date.now()}_${Math.random().toString(16).slice(2)}.${ext}`;
  ensureDir(uploadRoot);

  const filePath = path.join(uploadRoot, fileName);
  fs.writeFileSync(filePath, Buffer.from(base64, "base64"));

  return fileName; // store only filename; serve it via /uploads route
}

module.exports = { saveDataUrlImage };
