import React from "react";

export default function ItemCard({ item, onView }) {
  const imgSrc = item.imagePath ? `http://localhost:4000/uploads/${item.imagePath}` : null;

  return (
    <div className="card">
      <div className="row">
        <div className="thumb">
          {imgSrc ? <img src={imgSrc} alt={item.title} /> : <div className="thumb-placeholder">No Image</div>}
        </div>
        <div className="col">
          <div className="title">{item.title}</div>
          <div className="muted">{item.category || "Uncategorised"} • {item.itemCondition || "N/A"}</div>
          <div className="muted">Owner: {item.owner?.displayName}</div>
        </div>
        <div className="col right">
          <button onClick={() => onView(item.id)}>View</button>
        </div>
      </div>
    </div>
  );
}
